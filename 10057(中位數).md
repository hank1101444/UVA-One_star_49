## 題目: 
### output:
### 1.that line will contain the minimum possiblevalue for A. (v[temp])
### 2.Next it will contain how many numbers are there in the input that satisfy the property of A (The summation of absolute deviation from A is minimum).  (flag)
### 3.print how many possible different integer values are there for A (these values need not be present in the input).  (count) (多少不同數字)
## 注意 2222222, 中位數是2 , flag 要記得算到前面
```cpp
#include <iostream>
using namespace std;
#include <vector>
#include <algorithm>

int satisfy(vector <int> v, int index, int flag)
{
	for (int i = 0; i < v.size(); i++)
	{
		if (v[i] > v[index])
			break;
		else if (v[i] == v[index])
			flag++;
	}
	return flag;
}

int main()
{
	int n;
	vector <int> v;
	while (cin >> n)
	{
		while (n--)
		{
			int num;
			cin >> num;
			v.push_back(num);
		}
		sort(v.begin(), v.end());
		int temp, flag = 0;
		// temp, flag, count
		int count;
		if (v.size() % 2 == 1)
		{
			temp = v.size() / 2 + 1 - 1;
			flag = satisfy(v, temp, flag);
			count = 1;
		}

		else
		{
			temp = v.size() / 2 - 1;
			flag = satisfy(v, temp, flag);
			int temp2 = v.size() / 2 + 1 - 1;
			if (v[temp] != v[temp2])
				flag = satisfy(v, temp2, flag);
			count = v[temp2] - v[temp] + 1;
		}
		cout << v[temp] << ' ' << flag << ' ' << count << endl;
		v.clear();
	}


	return 0;
}
```
