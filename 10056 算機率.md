#include <iostream>
#include <cstdio>
using namespace std;
#include<cmath>

int main() {
	int n, people_num, i_win;
	double pro;
	cin >> n;
	while (n--) {
		cin >> people_num;
		cin >> pro;
		cin >> i_win;
		double result;
		if (pro <= 0.000001) {
			 result = 0.0000;
		}
		else {
			result = (pow((1 - pro), i_win - 1) * pro) * (1 / (1 - pow(1 - pro, people_num)));
		}
		printf("%.4lf", result);
		cout << endl;
	}

	return 0;
}
