#include <iostream>
using namespace std;
#include <cstring>
#include <string>

bool degree(int& a)
{

	int ans = 0;
	while (a != 0)
	{
		ans += a % 10;
		a /= 10;
	}
	a = ans;
	if (a % 9 == 0)
	{
		return true;
	}
	else
	{
		return false;
	}
}

int main()
{
	string s;
	while (cin >> s)
	{
		if (s == "0")
		{
			break;
		}
		cout << s;
		int temp = 0, times = 0;
		for (int i = 0; i < s.size(); i++)
		{
			temp += s[i] - '0';
		}
////////////////////////////////////////////////////////////////////////
		if (s.size() == 1) {
			if (degree(temp))
			{
				times++;
			}
		}
		else {
			while (temp / 10 != 0)
			{
				if (degree(temp))
				{
					times++;
				}
			}
			// while出來可能個位數是9
			if (degree(temp))
			{
				times++;
			}
		}
///////////////////////////////////////////////////////////////////////
		if (times != 0)
		{
			cout << " is a multiple of 9 and has 9-degree " << times << '.' << endl;
		}


		else
		{
			cout << " is not a multiple of 9." << endl;
		}

	}
	return 0;
}
