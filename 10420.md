# include <iostream>
using namespace std;
#include <vector>
#include <algorithm>
#include <string>

struct m{
	string key;
	int count = 1;
};

bool cmp (m a, m b) {
	//if (a.count != b.count)
	//	return a.count > b.count;
	//else
	// 他只要排字母大小
		return a.key < b.key;
}


int main() {

	int num;
	string country;
	string other;
	vector<m> data;
	cin >> num;

	while (num--) {
		
		cin >> country;

		//要加ignore 不然他只讀到country 不會讀到 other
		cin.ignore();
		getline(cin, other);
		int flag = 0;
		for (int i = 0; i < data.size(); i++) {
			if (country == data[i].key) {
				data[i].count++;
				break;
			}
			flag++;
		}
			
				
		if(flag == data.size())
		{
		m temp;
		temp.key = country;
		data.push_back(temp);
		}
	}
	// 依據vector裡的物件大小決定順序 會搬動整個 vector
	sort(data.begin(), data.end(), cmp);

	for (int i = 0; i < data.size(); i++) {
		cout << data[i].key << " " << data[i].count << endl;
	}


	return 0;
}
