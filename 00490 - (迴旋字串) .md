## 空白塞在'\0'前
## 注意有沒有讀到'\0'因為我是char array 每行後面要有結束字元

```cpp
#include <iostream>
using namespace std;
#include <string>
#include <vector>
// 直接vector <string> 就好....
struct z {
    string str;
};


int main() {
    // 要先將他都設為空字串 不然之後輸出會有問題
    char data[110][110];
    for (int i = 0; i < 110; i++) {
        for (int j = 0; j < 110; j++) {
            data[i][j] = ' ';
        }
    }
    vector <z> strdata;
    z a;
    int max = 0;
    while (getline(cin, a.str)) {
        if (a.str == "")
            break;
        if (a.str.size() > max)
            max = a.str.size();
        strdata.push_back(a);
    }
    int tempnum = 0;
    for (int i = strdata.size() - 1, k = 0; i >= 0; i--, k++) {
        int lastpointer = strdata[k].str.size();
        while (strdata[k].str.size() < tempnum) {
            strdata[k].str.insert(lastpointer, " ");
        }
        for (int j = 0; j <= strdata[k].str.size(); j++) {    // modify 要讀'\0'
            data[j][i] = strdata[k].str[j];
        }

        tempnum = strdata[k].str.size();
    }

    for (int i = 0; i < max; i++) {
        for (int j = 0; j < strdata.size(); j++) {
            cout << data[i][j];
        }
        cout << endl;
    }

    return 0;
}
```
