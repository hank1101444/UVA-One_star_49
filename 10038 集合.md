bool check(const vector <int> a, int n) {

	for (int i = 0; i < n - 1; i++) {
		if (a[i] != i + 1) {
			return false;
		}
	}
	return true;
}

void differ(vector <int>& a) {
	vector <int> temp;
	for (int i = 0; i < a.size() - 1; i++) {
		int difference = abs(a[i] - a[i + 1]);
		temp.push_back(difference);
	}
	if (temp.size() != 0) {
		a = temp;
	}
}
