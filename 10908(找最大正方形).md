# 注意邊界條件
```cpp
#include <iostream>
using namespace std;
#include <vector>
#include <string>
int main()
{
	vector <string>v;
	string s;
	int times;
	cin >> times;
	while (times--)
	{
		int m, n, q;
		cin >> m >> n >> q;
		cout << m << ' ' << n << ' ' << q << endl;
		cin.ignore();
		v.clear();
		for (int i = 0; i < m; i++)
		{
			getline(cin, s);
			v.push_back(s);
		}
		int n1, n2;
		for (int i = 0; i < q; i++)
		{
			cin >> n1 >> n2;
			int temp = 1;
			bool check = true;
			int size = 1, front, back;
			char flag = v[n1][n2];
      /////////////////////////////////////
			while (1)
			{
				// up down
				for (int i = n1 - temp; i <= n1 + temp; i++)
				{
          //注意是 >=
					if (i >= v.size() || i < 0)
						check = false;

					else {
						front = n2 - temp, back = n2 + temp;
						// front back
						for (int j = front; j <= back; j++)
						{
							if (v[i][j] != flag || j > v[i].size() || j < 0) {
								check = false;
								break;
							}
						}
					}
					if (!check)
						break;
				}
				if (check)
				{
					size += 2;
					temp++;
				}
				else
					break;
			}
			cout << size << endl;
		}
	}

	return 0;
}
```
