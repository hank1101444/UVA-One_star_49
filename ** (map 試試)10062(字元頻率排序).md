# eof 時不輸出 endl , 所以改成在下一筆測資開始時 output endl
```cpp
#include <iostream>
using namespace std;
#include <vector>
#include <algorithm>
struct z
{
    char ch;
    int times = 1;
};

bool cmp(z a, z b)
{
    if (a.times != b.times)
        return a.times < b.times;
    else
        return a.ch > b.ch;

}

bool check(vector <z>& v, char a)
{
    for (int i = 0; i < v.size(); i++)
    {
        if (v[i].ch == a)
        {
            v[i].times += 1;
            return true;
        }

    }
    return false;
}

int main()
{
    char a;
    vector <z> v;
    bool flag = false;
    while (cin.get(a))
    {
        if (a == '\n')
        {
            if (flag == false)
                flag = true;
            else
                cout << endl;
            sort(v.begin(), v.end(), cmp);
            for (int i = 0; i < v.size(); i++)
            {
                int temp = v[i].ch;
                cout << temp << ' ' << v[i].times << endl;
            }
            v.clear();
        }
        else
        {
            if (check(v, a)) {}
            else
            {
                z temp;
                temp.ch = a;
                v.push_back(temp);
            }
        }
    }

    return 0;
}
```
